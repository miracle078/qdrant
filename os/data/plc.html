⚙️ PLC-005: Data Layer Logic
Database management and query optimization control
Scan Time: 75ms | Tags: 198 | Routines: 22
📊 HMI Panel 📄 Area Index 🏠 Master PLC

Main Routine: DatabaseManager

(* Database Management Control Logic *) (* Handles connections, queries, and optimization *) (* Connection monitoring *) TokenDB_Connected := CheckConnection("os/data/sql/tokendb.sql"); QdrantDB_Connected := CheckConnection("http://localhost:6333"); IF NOT TokenDB_Connected THEN AlarmCode := 5001; (* Token DB connection lost *) PublishMQTT("chazon/os/data/error", AlarmCode); END_IF; IF NOT QdrantDB_Connected THEN AlarmCode := 5002; (* Qdrant DB connection lost *) PublishMQTT("chazon/os/data/error", AlarmCode); END_IF; (* Query processing *) IF CMD_ExecuteQuery AND TokenDB_Connected THEN QueryStartTime := SystemTime; QueryResult := ExecuteSQL(QueryString); QueryDuration := SystemTime - QueryStartTime; (* Update metrics *) TotalQueries := TotalQueries + 1; AvgQueryTime := (AvgQueryTime * (TotalQueries - 1) + QueryDuration) / TotalQueries; PublishMQTT("chazon/os/data/query/complete", QueryDuration); CMD_ExecuteQuery := FALSE; END_IF; (* Auto-backup logic *) IF ENABLE_AUTO_BACKUP THEN TimeSinceLastBackup := SystemTime - LastBackupTime; IF TimeSinceLastBackup > BackupInterval THEN CMD_BackupDatabase := TRUE; END_IF; END_IF; IF CMD_BackupDatabase THEN BackupResult := BackupDatabase("tokendb_backup"); IF BackupResult THEN LastBackupTime := SystemTime; PublishMQTT("chazon/os/data/backup/status", TRUE); ELSE AlarmCode := 5010; (* Backup failed *) PublishMQTT("chazon/os/data/error", AlarmCode); END_IF; CMD_BackupDatabase := FALSE; END_IF; (* Table optimization *) IF CMD_OptimizeTables THEN OptimizeAllTables(); PublishMQTT("chazon/os/data/optimize/complete", TRUE); CMD_OptimizeTables := FALSE; END_IF; (* Update Modbus registers *) ModbusHoldingReg[40050] := TotalRecords; ModbusHoldingReg[40051] := DatabaseSizeMB; ModbusInputReg[30050] := QueryRate;

Subroutine: ExecuteSQL

FUNCTION ExecuteSQL : STRING (* Execute SQL query with error handling *) VAR_INPUT query: STRING; END_VAR VAR result: STRING; errorCode: INT; END_VAR TRY (* Execute query using SQLite connection *) result := DB_Execute(TokenDB_Handle, query); (* Update record count if query modified data *) IF ContainsKeyword(query, "INSERT") OR ContainsKeyword(query, "UPDATE") OR ContainsKeyword(query, "DELETE") THEN TotalRecords := GetRecordCount(TokenDB_Handle); END_IF; ExecuteSQL := result; CATCH errorCode := 5020; (* Query execution failed *) PublishMQTT("chazon/os/data/error", errorCode); ExecuteSQL := "ERROR"; END_TRY; END_FUNCTION
