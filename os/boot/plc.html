⚙️ PLC-002: Boot Sequencer Logic
Sequential control with 6-phase state machine
Scan Time: 50ms | Tags: 156 | Routines: 18
📊 HMI Panel 📄 Area Index 🏠 Master PLC

Main Routine: Boot_Sequencer

(* Boot Sequencer State Machine *) (* Controls 6-phase initialization sequence *) CASE BootState OF 0: (* IDLE - Waiting for start command *) IF CMD_StartBoot THEN BootState := 1; CurrentPhase := 0; BootStartTime := SystemTime; STATUS_BootRunning := TRUE; END_IF; 1: (* LOADING - Execute current phase *) CASE CurrentPhase OF 0: PhaseComplete := LoadPhase0_Core(); 1: PhaseComplete := LoadPhase1_CLI(); 2: PhaseComplete := LoadPhase2_UI(); 3: PhaseComplete := LoadPhase3_Medical(); 4: PhaseComplete := LoadPhase4_Language(); 5: PhaseComplete := LoadPhase5_Programs(); END_CASE; IF PhaseComplete THEN PhaseCompleteBits[CurrentPhase] := TRUE; BootState := 2; END_IF; 2: (* TRANSITION - Move to next phase *) IF CurrentPhase < 5 THEN CurrentPhase := CurrentPhase + 1; BootState := 1; ELSE BootState := 3; (* All phases complete *) END_IF; 3: (* COMPLETE - Boot finished *) STATUS_BootRunning := FALSE; STATUS_BootComplete := TRUE; TotalBootTime := SystemTime - BootStartTime; PublishMQTT("chazon/os/boot/complete", TotalBootTime); 99: (* ERROR - Abort boot sequence *) STATUS_BootRunning := FALSE; STATUS_BootError := TRUE; PublishMQTT("chazon/os/boot/error", ErrorCode); END_CASE; (* Reset command *) IF CMD_ResetSequencer THEN BootState := 0; CurrentPhase := 0; PhaseCompleteBits := [FALSE, FALSE, FALSE, FALSE, FALSE, FALSE]; STATUS_BootComplete := FALSE; CMD_ResetSequencer := FALSE; END_IF;

Subroutine: LoadPhase0_Core

FUNCTION LoadPhase0_Core : BOOL (* Load core system modules *) VAR modules: ARRAY[0..2] OF STRING; i: INT; allLoaded: BOOL := TRUE; END_VAR modules[0] := "chazon-environment.md"; modules[1] := "chazon-mdcompiler.md"; modules[2] := "module-router.md"; FOR i := 0 TO 2 DO IF NOT LoadModule(modules[i]) THEN allLoaded := FALSE; END_IF; END_FOR; PublishMQTT("chazon/os/boot/phase/0/status", allLoaded); LoadPhase0_Core := allLoaded; END_FUNCTION
